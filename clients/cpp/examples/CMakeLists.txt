cmake_minimum_required(VERSION 3.20)
project(OrchidDBPublishedExample LANGUAGES CXX)
find_package(OrchidDB 0.4.0 CONFIG REQUIRED)
find_path(DUCKDB_INCLUDE_DIR duckdb.h REQUIRED)
find_library(DUCKDB_LIBRARY duckdb REQUIRED)
add_executable(orchiddb_example people.cpp)
target_compile_features(orchiddb_example PRIVATE cxx_std_17)
target_include_directories(orchiddb_example PRIVATE ${DUCKDB_INCLUDE_DIR})
target_link_libraries(orchiddb_example PRIVATE OrchidDB::orchiddb ${DUCKDB_LIBRARY})
enable_testing()
add_test(NAME published_client_example COMMAND orchiddb_example)
