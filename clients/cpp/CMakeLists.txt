cmake_minimum_required(VERSION 3.20)
project(OrchidDB VERSION 0.3.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(ORCHIDDB_BUILD_TESTS "Build integration tests (requires caller DuckDB library)" OFF)
option(ORCHIDDB_FETCH_JSON "Fetch pinned nlohmann_json if not installed" ON)
find_package(nlohmann_json 3.12 QUIET)
if(NOT nlohmann_json_FOUND)
  if(NOT ORCHIDDB_FETCH_JSON)
    message(FATAL_ERROR "nlohmann_json 3.12 required")
  endif()
  include(FetchContent)
  set(JSON_Install ON CACHE BOOL "" FORCE)
  FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(json)
endif()
add_library(orchiddb INTERFACE)
add_library(OrchidDB::orchiddb ALIAS orchiddb)
target_compile_features(orchiddb INTERFACE cxx_std_17)
target_include_directories(orchiddb INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(orchiddb INTERFACE nlohmann_json::nlohmann_json $<$<PLATFORM_ID:Linux>:dl>)
install(TARGETS orchiddb EXPORT OrchidDBTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/orchiddb)
set(ORCHIDDB_NATIVE_LIBRARY "" CACHE FILEPATH "Compiler shared library to package")
if(ORCHIDDB_NATIVE_LIBRARY)
  install(FILES "${ORCHIDDB_NATIVE_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
configure_package_config_file(cmake/OrchidDBConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/OrchidDBConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OrchidDB)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/OrchidDBConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT OrchidDBTargets NAMESPACE OrchidDB:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OrchidDB)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/OrchidDBConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/OrchidDBConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/OrchidDB)
if(ORCHIDDB_BUILD_TESTS)
  enable_testing()
  find_path(DUCKDB_INCLUDE_DIR duckdb.h REQUIRED)
  find_library(DUCKDB_LIBRARY duckdb REQUIRED)
  add_executable(orchiddb_integration tests/integration.cpp)
  target_link_libraries(orchiddb_integration PRIVATE OrchidDB::orchiddb ${DUCKDB_LIBRARY})
  target_include_directories(orchiddb_integration PRIVATE ${DUCKDB_INCLUDE_DIR})
  add_test(NAME integration COMMAND orchiddb_integration)
  add_executable(orchiddb_remote tests/remote.cpp)
  target_link_libraries(orchiddb_remote PRIVATE OrchidDB::orchiddb ${DUCKDB_LIBRARY})
  target_include_directories(orchiddb_remote PRIVATE ${DUCKDB_INCLUDE_DIR})
  add_test(NAME remote COMMAND orchiddb_remote)
endif()
set(CPACK_PACKAGE_FILE_NAME "orchiddb-cpp-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_GENERATOR TGZ)
include(CPack)
